S2Validator<JoinForm> JOIN = S2Validator.<JoinForm>builder()
    .field("name", "이름").rule(REQUIRED).rule(MAX_LENGTH, 20)
    .field("email", "이메일").rule(REQUIRED).rule(EMAIL)
    .field("mobile", "휴대폰 번호").rule(REQUIRED).rule(MPHONE_NO)
    .field("password", "비밀번호").rule(REQUIRED).rule(PASSWORD)
    .field("passwordConfirm", "비밀번호 확인").rule(REQUIRED).rule(EQUALS_FIELD, "password")
    .field("memberType", "회원 유형").rule(REQUIRED)
    .field("bizNo", "사업자등록번호").when("memberType", "BUSINESS")
        .rule(REQUIRED).rule(BIZRNO, true)                 // 사업자일 때만, 검증번호까지
    .field("interests", "관심사").rule(MIN_SIZE, 1).rule(MAX_SIZE, 3)
    .field("agree", "이용약관 동의").rule(ASSERT_TRUE)
    .build();

model.addAttribute("rules", JOIN.getRulesJson(locale));    // <form th:data-s2-rules="${rules}">
S2BindValidator.bind(JOIN).validate(form, bindingResult);  // POST: 같은 규칙으로 서버 검증

S2Validator<OrderForm> ORDER = S2Validator.<OrderForm>builder()
    .field("items[].name", "품목명").rule(REQUIRED)               // 모든 행에 적용
    .field("items[].qty", "수량").rule(REQUIRED).rule(MIN_VALUE, 1).rule(MAX_VALUE, 99)
    .field("startDate", "시작일").rule(REQUIRED).rule(DATE)
    .field("endDate", "종료일").rule(REQUIRED).rule(DATE).rule(DATE_AFTER, "startDate")
    .build();

// 행을 지우면 이름의 인덱스를 다시 매김: S2Validator.reindex(form, 'items')

public class ProfileForm {
    @NotBlank @Size(max = 10)  private String nickname;
    @NotNull @Min(14) @Max(120) private Integer age;
    @Pattern(regexp = "[0-9]{5}") private String zip;
    @Size(min = 1, max = 2)     private List<String> tags;
    @Past                       private LocalDate birthDate;   // 서버 전용
}

S2Validator<ProfileForm> rules = S2BeanValidation.of(ProfileForm.class)
    .labels(Map.of("nickname", "닉네임", "age", "나이", "zip", "우편번호", "tags", "태그"))
    .build();
model.addAttribute("rules", rules.getRulesJson(locale));   // 서버는 @Valid 그대로
